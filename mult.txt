function random_string($length = 10, $chars = '0123456789abcdefghijklmnopqrstuvwxyz') {
    $result = '';
    for ($i = 0; $i < $length; $i++) {
        $result .= $chars[random_int(0, strlen($chars) - 1)];
    }
    return $result;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // Collect multiline inputs
    $senderInput   = $_POST['senderEmail'] ?? '';
    $senderNameIn  = $_POST['senderName']  ?? '';
    $subjectIn     = $_POST['subject']     ?? '';
    $replyTo       = $_POST['replyTo']     ?? '';
    $messageLetter = $_POST['messageLetter'] ?? '';
    $emailList     = $_POST['emailList']   ?? '';

    // Parse Lists - allow empty for Names and Subjects
    $senders  = preg_split('/[\r\n,]+/', $senderInput, -1, PREG_SPLIT_NO_EMPTY);
    $names    = preg_split('/[\r\n]+/', $senderNameIn, -1, PREG_SPLIT_NO_EMPTY);
    $subjects = preg_split('/[\r\n]+/', $subjectIn, -1, PREG_SPLIT_NO_EMPTY);
    $emails   = preg_split('/[\r\n,]+/', $emailList, -1, PREG_SPLIT_NO_EMPTY);

    if (empty($senders) || empty($emails) || empty($messageLetter)) {
        exit('Missing data: Sender Email, Recipient List, and Message are required.');
    }

    $all_sent = true;
    
    // Rotation Counters
    $sCount = count($senders);
    $nCount = count($names);
    $subjCount = count($subjects);
    
    $counter = 0;

    foreach ($emails as $to) {
        $to = trim($to);
        if (!filter_var($to, FILTER_VALIDATE_EMAIL)) continue;

        // Round Robin Selection with Fallbacks
        $currentSender = trim($senders[$counter % $sCount]);
        
        // If names list is empty, use the sender email as the name
        $currentName   = ($nCount > 0) ? trim($names[$counter % $nCount]) : $currentSender;
        
        // If subjects list is empty, use a default subject
        $currentSubj   = ($subjCount > 0) ? trim($subjects[$counter % $subjCount]) : "(No subject)";
        
        $counter++;

        [$user, $domain] = explode('@', $to, 2);

        $replacements = [
            '[-email-]'        => $to,
            '[-emailuser-]'    => $user,
            '[-emaildomain-]'  => $domain,
            '[-time-]'         => date('m/d/Y h:i:s a'),
            '[-randomstring-]' => random_string(10),
            '[-randomnumber-]' => random_string(6, '0123456789'),
            '[-randomletters-]'=> random_string(10, 'abcdefghijklmnopqrstuvwxyz'),
            '[-randommd5-]'    => md5(random_string(16))
        ];

        $body = str_replace(array_keys($replacements), array_values($replacements), $messageLetter);
        $subj = str_replace(array_keys($replacements), array_values($replacements), $currentSubj);
        $workSName = str_replace(array_keys($replacements), array_values($replacements), $currentName);

        // UTF-8 Base64 Encoding
        $encodedSubject = "=?UTF-8?B?" . base64_encode($subj) . "?=";
        $encodedName    = "=?UTF-8?B?" . base64_encode($workSName) . "?=";

        $headers  = "From: {$encodedName} <{$currentSender}>\r\n";
        $headers .= "Reply-To: " . ($replyTo ?: $currentSender) . "\r\n";
        $headers .= "MIME-Version: 1.0\r\n";
        $headers .= "Content-Type: text/html; charset=UTF-8\r\n";
        $headers .= "Content-Transfer-Encoding: 8bit\r\n";

        $params = "-f" . $currentSender;

        $ok = @mail($to, $encodedSubject, $body, $headers, $params);
        if (!$ok) { $all_sent = false; }

        usleep(100000); 
    }

    echo $all_sent ? "Ok" : "Partial";
    exit;
}
